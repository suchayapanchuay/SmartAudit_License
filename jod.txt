# routes/order_route.py
from datetime import datetime
import json
import random
import string
from typing import Any, Dict, List, Optional

from fastapi import APIRouter, Depends, HTTPException, Response
from pydantic import BaseModel, EmailStr, Field
from sqlalchemy.orm import Session

from database import get_db
from models.customer import Customer
from models.product import Product
from models.order import Order

# publish เป็น no-op ถ้าไม่มี utils.events
try:
    from utils.events import publish  # type: ignore
except Exception:  # pragma: no cover
    def publish(event: str, payload: Dict[str, Any]) -> None:
        pass

router = APIRouter(prefix="/api", tags=["orders"])

# ---------- Schemas ----------
class OrderItemIn(BaseModel):
    sku: Optional[str] = None
    name: Optional[str] = None
    qty: int = Field(default=1, ge=1)
    price: int = Field(default=0, ge=0)

class OrderIn(BaseModel):
    customer_name: Optional[str] = None
    customer_email: EmailStr
    company: Optional[str] = None
    phone: Optional[str] = None
    items: List[OrderItemIn] = []       # ← อนุญาตว่าง แล้วค่อยเติมให้
    grand_total: int = Field(default=0, ge=0)
    note: Optional[str] = None
    form_type: Optional[str] = None

class OrderOut(BaseModel):
    id: int
    order_code: str
    customer_email: EmailStr
    customer_name: Optional[str]
    company: Optional[str]
    phone: Optional[str]
    items: List[Dict[str, Any]]
    grand_total: int
    currency: str
    status: str
    note: Optional[str]
    created_at: Optional[str]

def _code() -> str:
    return "ORD-" + "".join(random.choices(string.digits, k=10))

def _safe_json_loads(s: Optional[str]) -> Dict[str, Any]:
    try:
        return json.loads(s or "{}")
    except Exception:
        return {}

def _to_out(o: Order) -> OrderOut:
    meta = _safe_json_loads(o.meta)
    return OrderOut(
        id=o.id,
        order_code=o.order_code,
        customer_email=meta.get("customer_email"),
        customer_name=meta.get("customer_name"),
        company=meta.get("company"),
        phone=meta.get("phone"),
        items=meta.get("items", []),
        grand_total=o.amount_cents,
        currency=o.currency or "THB",
        status=o.status,
        note=meta.get("note"),
        created_at=o.created_at.isoformat() if o.created_at else None,
    )

# ---------- Diagnostics ----------
@router.get("/orders/_diag")
def orders_diag():
    return {
        "handler": "v2",
        "auto_fill_items_if_empty": True,
        "build": "2025-11-03",
    }

# ---------- Handlers ----------
@router.post("/orders", response_model=OrderOut, status_code=201)
def create_order_v2(body: OrderIn, db: Session = Depends(get_db), response: Response = None):
    """
    v2 handler: auto-fill items if empty; compatible with models.product.Product (ไม่มีฟิลด์ term/duration/max_activations)
    """
    # (A) เซ็น header เพื่อวินิจฉัยว่าใช้ v2 จริง
    if response is not None:
        response.headers["x-order-handler"] = "v2"

    # (B) ถ้า items ว่าง ให้เติม Lead item อัตโนมัติ
    if not body.items:
        body.items = [OrderItemIn(sku="CONTACT_SALES", name="Contact Sales Lead", qty=1, price=0)]

    first = body.items[0]

    # (C) upsert customer
    cust = db.query(Customer).filter(Customer.email == str(body.customer_email)).first()
    if not cust:
        cust = Customer(
            email=str(body.customer_email),
            name=(body.customer_name or None),
            created_at=datetime.utcnow(),
        )
        db.add(cust)
        db.flush()
    else:
        if body.customer_name and body.customer_name.strip() and cust.name != body.customer_name.strip():
            cust.name = body.customer_name.strip()

    # (D) find/create product
    prod = None
    if first.sku:
        prod = db.query(Product).filter(Product.sku == first.sku).first()
    if not prod:
        gen_sku = first.sku or ("SKU" + "".join(random.choices(string.ascii_uppercase + string.digits, k=8)))
        gen_name = first.name or "Custom Product"
        prod = Product(
            sku=gen_sku,
            name=gen_name,
            category="lead",
            is_active=True,
            description="Auto-created from lead form",
            meta={"source": "lead_form"},
            created_at=datetime.utcnow(),
        )
        db.add(prod)
        db.flush()

    # (E) amount
    amount_cents = int(body.grand_total or 0)
    if amount_cents == 0:
        try:
            amount_cents = int(first.price or 0) * int(first.qty or 1)
        except Exception:
            amount_cents = 0

    # (F) create order
    meta = {
        "customer_email": str(body.customer_email),
        "customer_name": body.customer_name,
        "company": body.company,
        "phone": body.phone,
        "items": [x.model_dump() for x in body.items],
        "grand_total": body.grand_total,
        "note": body.note,
        "form_type": body.form_type,
        "source": "public_site",
    }

    o = Order(
        order_code=_code(),
        customer_id=cust.id,
        product_id=prod.id,
        amount_cents=amount_cents,
        currency="THB",
        status="pending",
        meta=json.dumps(meta, ensure_ascii=False),
        created_at=datetime.utcnow(),
    )

    db.add(o)
    db.commit()
    db.refresh(o)

    publish("order_created", {
        "id": o.id,
        "order_code": o.order_code,
        "customer_email": body.customer_email,
        "customer_name": body.customer_name,
        "product_sku": prod.sku,
        "amount_cents": o.amount_cents,
        "created_at": o.created_at.isoformat() if o.created_at else None,
    })

    return _to_out(o)
